function aplicarFiltrosRelatorio() { let stFiltro = document.getElementById('filtroTipo').value.toLowerCase(); let txtBusca = document.getElementById('filtroBusca').value.trim().toLowerCase(); let dadosFiltrados = parceirosCache.filter(p => { let tipoParc = (p.tipo_cadastro || 'cliente').toLowerCase(); // Filtro exato por tipo de relacionamento if (stFiltro !== 'todos') { if (stFiltro === 'cliente' && !tipoParc.includes('cliente')) return false; if (stFiltro === 'fornecedor' && !tipoParc.includes('fornecedor')) return false; if (stFiltro === 'ambos' && !tipoParc.includes('ambos') && !tipoParc.includes('&')) return false; } if (txtBusca) { let linha = (p.nome + ' ' + (p.cnpj_cpf || '') + ' ' + (p.cidade_uf || '') + ' ' + (p.centro_custo || '')).toLowerCase(); if (!linha.includes(txtBusca)) return false; } return true; }); const tbodyRel = document.getElementById('tbodyRelatorioModo'); tbodyRel.innerHTML = ''; if(dadosFiltrados.length === 0) { tbodyRel.innerHTML = `Nenhum registro encontrado.`; return; } dadosFiltrados.forEach(p => { let badge = ''; let tipo = (p.tipo_cadastro || 'Cliente').toLowerCase(); if(tipo.includes('fornecedor') && !tipo.includes('ambos') && !tipo.includes('&')) badge = 'Fornecedor'; else if(tipo.includes('ambos') || tipo.includes('&')) badge = 'Cliente & Forn.'; else badge = 'Cliente'; let enderecoCompleto = `${p.logradouro || ''}, nº ${p.numero || ''} - ${p.bairro || ''} - ${p.cidade_uf || ''} (CEP: ${p.cep || '-'})`; tbodyRel.innerHTML += ` ${badge} ${p.cnpj_cpf || '-'} ${p.nome || '-'} ${p.centro_custo || '-'} Contato: ${p.nome_contato || '-'} ${p.email || '-'}
${p.telefone || '-'} ${enderecoCompleto} `; }); }
